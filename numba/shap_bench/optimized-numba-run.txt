Exact verification/benchmark script (retained here for reproducibility):

from itertools import product
import platform
import statistics
import time

import llvmlite
from llvmlite import binding as llvm
import numba
import numpy as np

from bench_compiler_fusion import load_kernels, SOURCE_SHA256
from optimized_numba import _delta_minimization_order as optimized
from optimized_numba import _reverse_window_score_gain as score


assert not numba.config.ENABLE_PROFILING
assert not numba.config.DEBUGINFO_DEFAULT
originals = load_kernels()
original = originals['_delta_minimization_order']
print(f'Python {platform.python_version()} | {platform.platform()}')
print(f'NumPy {np.__version__} | Numba {numba.__version__} | '
      f'llvmlite {llvmlite.__version__} | LLVM {llvm.llvm_version_info}')
print(f'Numba: {numba.__file__}')
print(f'Original SHA256: {SOURCE_SHA256}')

gains = set()
for bits in product((False, True), repeat=4):
    masks = np.array(bits, dtype=np.bool_).reshape(4, 1)
    order = np.arange(4)
    expected = originals['_reverse_window_score_gain'](masks, order, 1, 2)
    assert score(masks, order, 1, 2) == expected
    assert score.py_func(masks, order, 1, 2) == expected
    gains.add(expected)
assert gains == {-2, 0, 2}

rng = np.random.default_rng(51)
cases = 0
for shape in ((0, 0), (1, 0), (3, 0), (4, 1), (5, 3), (16, 5), (256, 8)):
    for base in (np.zeros(shape, dtype=np.bool_), np.ones(shape, dtype=np.bool_),
                 rng.integers(0, 2, shape).astype(np.bool_)):
        for masks in (base, base[:, ::-1], base[::2], np.asfortranarray(base)):
            unchanged = masks.copy()
            for window, passes in ((0, 0), (2, 2), (3, 1), (9, 2),
                                   (100, 2), (-2, 1), (100, -1)):
                expected = original(masks, window, passes)
                result = optimized(masks, window, passes)
                np.testing.assert_array_equal(result, expected)
                np.testing.assert_array_equal(np.sort(result), np.arange(len(masks)))
                np.testing.assert_array_equal(masks, unchanged)
                cases += 1
print(f'PASS: {cases} exact-order/input-preservation cases + 16 signed-gain cases')

for signature in score.signatures:
    module = llvm.parse_assembly(score.inspect_llvm(signature))
    native = str(module.get_function(score.overloads[signature].fndesc.mangled_name))
    assert 'NRT_MemInfo_alloc' not in native
    assert 'NRT_incref' not in native and 'NRT_decref' not in native
print(f'PASS: scoring native LLVM has no NRT allocations/refcount calls '
      f'({len(score.signatures)} signatures)')

for features in (8, 13):
    masks = np.random.default_rng(0).integers(
        0, 2, (2**features, features)).astype(np.bool_)
    expected = original(masks)
    functions = {'original': original, 'optimized': optimized}
    for function in functions.values():
        for _ in range(2):
            np.testing.assert_array_equal(function(masks), expected)
    samples = {name: [] for name in functions}
    for repeat in range(10):
        names = list(functions)
        if repeat % 2:
            names.reverse()
        for name in names:
            start = time.perf_counter_ns()
            result = functions[name](masks)
            elapsed = (time.perf_counter_ns() - start) / 1e6
            samples[name].append(elapsed)
            np.testing.assert_array_equal(result, expected)
    print(f'features={features}, masks={len(masks)}, n=10, milliseconds')
    for name, values in samples.items():
        print(f'{name}: mean={statistics.mean(values):.4f}, '
              f'std={statistics.stdev(values):.4f}, '
              f'median={statistics.median(values):.4f}')
        print(f'{name} raw: {values}')
    print(f'median speedup: '
          f'{statistics.median(samples["original"]) / statistics.median(samples["optimized"]):.2f}x')
